from scapy.all import *

SECRET_HEADER = "X-Secret-Data"
received = {}

def assemble(pkt):
    global received
    if pkt.haslayer(Raw):
        payload = pkt[Raw].load.decode(errors='ignore')
        
        if SECRET_HEADER in payload:
            # Витягуємо тільки потрібний нам рядок
            try:
                content = payload.split(f"{SECRET_HEADER}: ")[1].split("\r\n")[0]
                idx, total, text = content.split(":", 2)
                
                idx = int(idx)
                if idx not in received:
                    received[idx] = text
                    print(f"\n[!!!] ПЕРЕМОГА! Отримано шматок {idx}: {text}")
                    
                    if len(received) == int(total):
                        res = "".join([received[i] for i in sorted(received.keys())])
                        print(f"ЗІБРАНО: {res}")
                        with open("received_secret.txt", "w") as f: f.write(res)
            except:
                pass

print("[*] Асемблер чекає тільки на секретні дані...")
sniff(filter="tcp port 80", prn=assemble, store=0)
